% Build with latexmk from this directory. The PDF is written to build/main.pdf.
\documentclass[a4paper,12pt]{article}

% =========================
% Layout and paragraph style
% =========================
\usepackage[left=3.18cm,right=3.18cm,top=2.54cm,bottom=2.54cm]{geometry}

% Use the original report fonts when installed, with TeX Live alternatives.

  {\setmainfont{Times New Roman}}
  {\setmainfont{texgyretermes}[Extension=.otf,
     UprightFont=*-regular, BoldFont=*-bold,
     ItalicFont=*-italic, BoldItalicFont=*-bolditalic]}

  {\setCJKmainfont{\ReportCJKFont}[Script=Default]}
  {\setCJKmainfont{FandolSong-Regular.otf}[Script=Default,
     BoldFont=FandolSong-Bold.otf, ItalicFont=FandolKai-Regular.otf]}

% Reuse the configured CJK font in sans and monospaced labels.
\renewcommand*{\CJKsfdefault}{\CJKrmdefault}
\renewcommand*{\CJKttdefault}{\CJKrmdefault}
